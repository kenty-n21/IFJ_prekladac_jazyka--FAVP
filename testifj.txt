# Referencny priklad pre: if (x == 0) { y = 1 }

DEFVAR GF@x
DEFVAR GF@y

# Pociatocne hodnoty: x = 0, y = 0
MOVE GF@x int@0
MOVE GF@y int@0

# Podmienka: ak x != 0, preskoc telo if na navestie endif
JUMPIFNEQ endif GF@x int@0

# Telo podmienky: y = 1
MOVE GF@y int@1

LABEL endif

# Vypis vysledku (vypise 1)
WRITE GF@y
